    Extractors

    LLM-based extractors. See the Information extraction quick start for a walkthrough.

    LLM-based entity extractors.

    SimpleLLMExtractor extracts entities described by EntityDefinitions from clinical documents with one structured-output LLM call per document, then validates the result with aatm.data_models.extractors.validate_extracted_entity.

    Extraction is asynchronous; await the extractor (or use asyncio.run in scripts):

    from aatm.data_models.extractors import EntityDefinitions
from aatm.extractors import SimpleLLMExtractor

extractor = SimpleLLMExtractor(
    model_id="openai/gpt-5",  # or "gemini/...", "anthropic/...", "ollama/...", ...
    entity_definitions=EntityDefinitions.from_predefined("condition_occurrence"),
)
results = await extractor([clinical_note])
records = results[0].to_records()

    Model ids are prefixed with a provider; see aatm.llm.create_llm for the list. The API key is read from the provider's usual environment variable unless passed explicitly.

    class SimpleLLMExtractor:
    """Extract entities from texts with a single structured-output LLM call per text.

    The text is split into numbered lines (`segment_text`) before being sent to the
    model, which must return the line id of each mention. The output is validated with
    `validate_extracted_entity`, which grounds each span to one position on its line and
    checks keys and parent references.

    Args:
        model_id: Provider-prefixed model id, e.g. ``"openai/gpt-5"``,
            ``"gemini/gemini-3-flash-preview"``, ``"anthropic/claude-opus-5-5"`` or
            ``"ollama/llama3.3"``. See `aatm.llm.create_llm`. Ignored when ``llm`` is
            given.
        entity_definitions: Definitions of the entities to extract.
        instructions: System prompt template. The placeholders ``{output_format}`` and
            ``{entity_definitions}`` are replaced; other braces are kept as-is.
        output_type: Structured output model requested from the LLM.
        api_key: Optional API key. If omitted, the provider's default environment
            variable (e.g. ``OPENAI_API_KEY``) is used.
        base_url: Server URL for OpenAI-compatible providers (vLLM, Ollama, ...).
        settings: Generation settings, as an `aatm.llm.LLMSettings` or a compatible
            dict, e.g. ``{"temperature": 0, "max_tokens": 8000}``.
        max_concurrency: Maximum number of concurrent model calls made by this
            extractor. ``None`` (default) sets no cap: all texts are sent at once,
            within any limits set for the model with `aatm.llm.set_rate_limits`
            (none by default). Rate-limit and transient errors are retried either way.
        max_segment_chars: If set, lines longer than this are split at sentence
            boundaries into separately numbered segments before being sent to the
            model. ``None`` (default) keeps every line whole.
        llm: A ready-made `aatm.llm.StructuredLLM` to use instead of building one
            from ``model_id``, e.g. a custom provider.
        on_error: What to do when a text cannot be extracted once retries are spent:
            ``"raise"`` (default) raises the error; ``"record"`` returns an empty
            result for that text with the error in its ``error`` field, so one note
            does not stop a batch.
    """

    def __init__(
        self,
        model_id: Optional[str] = None,
        entity_definitions: Optional[EntityDefinitions] = None,
        instructions: str = DEFAULT_INSTRUCTION,
        output_type: type[ExtractedEntityCandidates] = ExtractedEntityCandidates,
        api_key: Optional[str] = None,
        base_url: Optional[str] = None,
        settings: Optional[LLMSettings | dict[str, Any]] = None,
        max_concurrency: Optional[int] = None,
        max_segment_chars: Optional[int] = None,
        llm: Optional[StructuredLLM] = None,
        on_error: Literal["raise", "record"] = "raise",
    ):
        if entity_definitions is None:
            raise ValueError("entity_definitions is required")
        if on_error not in ("raise", "record"):
            raise ValueError("on_error must be 'raise' or 'record'")
        if max_concurrency is not None and max_concurrency < 1:
            raise ValueError("max_concurrency must be >= 1 or None")
        if llm is None:
            if model_id is None:
                raise ValueError("Either model_id or llm must be given")
            llm = create_llm(
                model_id, api_key=api_key, base_url=base_url, settings=settings
            )

        self.llm = llm
        self.model_id = llm.model_id
        self.max_concurrency = max_concurrency
        self.semaphore: Optional[asyncio.Semaphore] = (
            asyncio.Semaphore(max_concurrency) if max_concurrency is not None else None
        )
        self.on_error = on_error
        self.entity_definitions = entity_definitions
        self.max_segment_chars = max_segment_chars

        # Plain replacement instead of str.format so that literal braces in custom
        # instructions do not break prompt construction.
        self.instructions = instructions.replace(
            "{output_format}", str(output_type.model_json_schema())
        ).replace("{entity_definitions}", entity_definitions.to_prompt_str())
        self.output_type = output_type

    async def extract(self, text: str) -> EntityValidationResult:
        segments = segment_text(text, max_segment_chars=self.max_segment_chars)

        try:
            async with self.semaphore or contextlib.nullcontext():
                candidates = await self.llm.generate(
                    system=self.instructions,
                    user=format_segments_for_prompt(segments),
                    output_type=self.output_type,
                )
        except Exception as e:
            if self.on_error == "raise":
                raise
            logger.warning("Extraction failed with %s: %s", self.model_id, e)
            return EntityValidationResult(error=f"{type(e).__name__}: {e}")

        return validate_extracted_entity(
            text=text,
            extracted_entities=candidates,
            entity_definitions=self.entity_definitions,
            segments=segments,
        )

    # --- provider batch jobs ------------------------------------------------------

    @property
    def supports_batch(self) -> bool:
        """Whether the model can run provider batch jobs (OpenAI, Anthropic, Gemini)."""
        return supports_batch(self.llm)

    def _batch_llm(self) -> BatchLLM:
        if not supports_batch(self.llm):
            raise ValueError(
                f"{self.model_id} has no batch endpoint here: batch extraction "
                "supports the openai, anthropic and gemini providers"
            )
        return self.llm  # type: ignore[return-value]

    async def submit_batch(self, items: Sequence[tuple[str, str]]) -> str:
        """Submit texts as one provider batch job; return the job id.

        Args:
            items: ``(key, text)`` pairs. Keys are unique within the batch, made of
                letters, digits, ``_`` and ``-``.
        """
        requests = [
            BatchRequest(
                custom_id=key,
                system=self.instructions,
                user=format_segments_for_prompt(
                    segment_text(text, max_segment_chars=self.max_segment_chars)
                ),
            )
            for key, text in items
        ]
        return await self._batch_llm().submit_batch(requests, self.output_type)

    async def poll_batch(
        self, batch_id: str, items: Sequence[tuple[str, str]]
    ) -> Optional[dict[str, EntityValidationResult]]:
        """``None`` while the job runs; then the validated result of each key.

        `items` are the ``(key, text)`` pairs submitted. A request that failed gets
        a result with its ``error`` set, whatever ``on_error`` is.
        """
        keys = [key for key, _ in items]
        batch = await self._batch_llm().poll_batch(batch_id, keys, self.output_type)
        if batch is None:
            return None
        results: dict[str, EntityValidationResult] = {}
        for (key, text), item in zip(items, batch):
            if item.error is not None or item.output is None:
                results[key] = EntityValidationResult(error=item.error or "no output")
                continue
            try:
                results[key] = validate_extracted_entity(
                    text=text,
                    extracted_entities=item.output,
                    entity_definitions=self.entity_definitions,
                    segments=segment_text(
                        text, max_segment_chars=self.max_segment_chars
                    ),
                )
            except Exception as e:  # noqa: BLE001 - one bad output must not stop a batch
                results[key] = EntityValidationResult(error=f"{type(e).__name__}: {e}")
        return results

    async def __call__(self, texts: List[str]) -> List[EntityValidationResult]:
        if not isinstance(texts, list) or not all(isinstance(t, str) for t in texts):
            raise ValueError("Input must be a list of strings.")

        if not texts:
            return []

        return list(await asyncio.gather(*(self.extract(t) for t in texts)))

    LLM providers

    Provider-neutral structured-output LLM clients.

    from aatm.llm import create_llm

llm = create_llm("anthropic/claude-opus-5-5", settings={"reasoning_effort": "low"})
result = await llm.generate(system="...", user="...", output_type=MyModel)

    Supported prefixes: openai, gemini, anthropic (needs aatm[anthropic]), ollama, openrouter and openai-compatible (any OpenAI-compatible server, with base_url). More can be added with register_provider.

    Decision models answer closed-set questions (choice, score, yes/no) with probabilities instead of writing text: see aatm.llm.decisions and create_decision_model.

    Calls are retried on rate limits and transient errors. No rate limits are set by default; set them, per model, provider or for all, with set_rate_limits (see aatm.llm.limits).

    class RateLimiter:
    """Concurrency cap, sliding-window rate limits and retries for one model.

    Get one with `get_rate_limiter`; its limits follow `set_rate_limits`. The windows
    are shared by every event loop; the concurrency cap applies per event loop.
    """

    def __init__(self, model_id: str, limits: RateLimits):
        self.model_id = model_id
        self.limits = limits
        self._lock = threading.Lock()
        self._requests: deque[float] = deque()
        self._tokens: deque[list[float]] = deque()  # [start time, tokens]
        self._paused_until = 0.0
        self._loops: weakref.WeakKeyDictionary[
            asyncio.AbstractEventLoop, _LoopState
        ] = weakref.WeakKeyDictionary()

    # --- waiting ------------------------------------------------------------------

    def pause(self, seconds: float) -> None:
        """Hold every new call to this model for `seconds`, e.g. after a 429."""
        with self._lock:
            self._paused_until = max(self._paused_until, time.monotonic() + seconds)

    def _wait_seconds(self, now: float, tokens: int) -> float:
        """Seconds to wait before a call may start; 0 reserves it. Lock held."""
        limits = self.limits
        wait = self._paused_until - now
        horizon = now - WINDOW_SECONDS
        while self._requests and self._requests[0] <= horizon:
            self._requests.popleft()
        while self._tokens and self._tokens[0][0] <= horizon:
            self._tokens.popleft()
        rpm = limits.requests_per_minute
        if rpm is not None and len(self._requests) >= rpm:
            index = int(len(self._requests) - rpm)
            wait = max(wait, self._requests[index] + WINDOW_SECONDS - now)
        tpm = limits.tokens_per_minute
        if tpm is not None and self._tokens:
            excess = sum(t for _, t in self._tokens) + tokens - tpm
            if excess > 0:
                # Wait until enough of the oldest reservations leave the window.
                freed = 0.0
                for started, used in self._tokens:
                    freed += used
                    if freed >= excess:
                        break
                wait = max(wait, started + WINDOW_SECONDS - now)
        return wait

    async def _reserve(self, tokens: int) -> Optional[list[float]]:
        while True:
            with self._lock:
                now = time.monotonic()
                wait = self._wait_seconds(now, tokens)
                if wait <= 0:
                    if self.limits.requests_per_minute is not None:
                        self._requests.append(now)
                    if self.limits.tokens_per_minute is None:
                        return None
                    entry = [now, float(tokens)]
                    self._tokens.append(entry)
                    return entry
            await asyncio.sleep(wait)

    def _loop_state(self) -> _LoopState:
        loop = asyncio.get_running_loop()
        state = self._loops.get(loop)
        if state is None:
            state = self._loops[loop] = _LoopState()
        return state

    @asynccontextmanager
    async def slot(self, estimated_tokens: int = 0) -> AsyncIterator[_Slot]:
        """Wait for a concurrency slot and for room in the rate windows."""
        state = self._loop_state()
        async with state.condition:
            await state.condition.wait_for(
                lambda: (
                    self.limits.max_concurrency is None
                    or state.active < self.limits.max_concurrency
                )
            )
            state.active += 1
        try:
            entry = await self._reserve(estimated_tokens)
            yield _Slot(self, entry)
        finally:
            async with state.condition:
                state.active -= 1
                state.condition.notify_all()

    def backoff(self, attempt: int) -> float:
        """Random wait before retry number `attempt` (0-based): full jitter."""
        cap = min(self.limits.max_backoff, self.limits.initial_backoff * 2**attempt)
        return random.uniform(0, cap)

    def add(
        self,
        input_tokens: Optional[int],
        output_tokens: Optional[int],
        model_id: Optional[str] = None,
        cost: Optional[float] = None,
    ) -> None:
        with _LOCK:
            model = self.by_model.get(model_id or UNKNOWN_MODEL)
            if model is None:
                model = self.by_model[model_id or UNKNOWN_MODEL] = ModelTokens()
            for counts in (self, model):
                counts.calls += 1
                counts.input_tokens += input_tokens or 0
                counts.output_tokens += output_tokens or 0
                counts.reported_cost += cost or 0.0

    def merge(self, other: "TokenUsage") -> None:
        """Add the counts of another tracker to this one."""
        with _LOCK:
            self.calls += other.calls
            self.input_tokens += other.input_tokens
            self.output_tokens += other.output_tokens
            self.reported_cost += other.reported_cost
            for model_id, counts in other.by_model.items():
                mine = self.by_model.setdefault(model_id, ModelTokens())
                mine.calls += counts.calls
                mine.input_tokens += counts.input_tokens
                mine.output_tokens += counts.output_tokens
                mine.reported_cost += counts.reported_cost

    def cost(self, prices: Optional[Prices]) -> Optional[float]:
        """The estimated cost of the calls, or ``None`` when nothing gives a cost.

        Each model costs its tokens at its price in `prices` or, without one, the
        cost its provider reported (`ModelTokens.reported_cost`). Models with
        neither are left out: see `unpriced`. Batch calls (``<model>@batch``) cost
        the model's price times `BATCH_DISCOUNT`, unless `prices` has the ``@batch``
        id. Returns ``None`` when no prices are given and no provider reported a
        cost.
        """
        normalized = model_prices(prices)
        if not normalized and not self.reported_cost:
            return None
        total = 0.0
        for model_id, counts in self.by_model.items():
            price = price_of(model_id, normalized)
            if price is not None:
                total += counts.cost_at(price)
            else:
                total += counts.reported_cost
        return total

    def unpriced(self, prices: Optional[Prices]) -> list[str]:
        """The models called that have no price in `prices` and reported no cost."""
        normalized = model_prices(prices)
        return sorted(
            m
            for m, counts in self.by_model.items()
            if price_of(m, normalized) is None and not counts.reported_cost
        )

    Rate limits and retries

    Optional rate limits, and retries, for every LLM call.

    Every call made through the adapters in aatm.llm goes through the RateLimiter of its model, shared by all the components of a pipeline (extractor, concept search agent, selectors...).

    No limits are set by default: calls run as fast and as concurrently as the code issues them. Set limits when a provider's quota requires it, or to keep long runs under it, once for the whole process:

    from aatm.llm import set_rate_limits

set_rate_limits("*", max_concurrency=16)  # every model
set_rate_limits("anthropic", requests_per_minute=1000, tokens_per_minute=400_000)
set_rate_limits("openai/gpt-5-mini", max_concurrency=64)  # one model

    Limits are looked up by model id ("provider/model"), then provider, then "*". Each model has its own limiter, even when its limits come from the provider entry. Set them before or during a run; models already in use pick them up on their next call.

    Retries are on by default. Rate-limit errors (429), server errors (5xx), timeouts and connection errors are retried up to 6 times with exponential backoff and full jitter, honouring Retry-After when the provider sends it. A 429 also pauses every call to that model for the wait, so concurrent calls don't keep hitting the limit. Other errors are raised at once. Turn retries off with max_retries=0.

        self._tokens: deque[list[float]] = deque()  # [start time, tokens]
        self._paused_until = 0.0
        self._loops: weakref.WeakKeyDictionary[
            asyncio.AbstractEventLoop, _LoopState
        ] = weakref.WeakKeyDictionary()

    # --- waiting ------------------------------------------------------------------

    def pause(self, seconds: float) -> None:
        """Hold every new call to this model for `seconds`, e.g. after a 429."""
        with self._lock:
            self._paused_until = max(self._paused_until, time.monotonic() + seconds)

    def _wait_seconds(self, now: float, tokens: int) -> float:
        """Seconds to wait before a call may start; 0 reserves it. Lock held."""
        limits = self.limits
        wait = self._paused_until - now
        horizon = now - WINDOW_SECONDS
        while self._requests and self._requests[0] <= horizon:
            self._requests.popleft()
        while self._tokens and self._tokens[0][0] <= horizon:
            self._tokens.popleft()
        rpm = limits.requests_per_minute
        if rpm is not None and len(self._requests) >= rpm:
            index = int(len(self._requests) - rpm)
            wait = max(wait, self._requests[index] + WINDOW_SECONDS - now)
        tpm = limits.tokens_per_minute
        if tpm is not None and self._tokens:
            excess = sum(t for _, t in self._tokens) + tokens - tpm
            if excess > 0:
                # Wait until enough of the oldest reservations leave the window.
                freed = 0.0
                for started, used in self._tokens:
                    freed += used
                    if freed >= excess:
                        break
                wait = max(wait, started + WINDOW_SECONDS - now)
        return wait

    async def _reserve(self, tokens: int) -> Optional[list[float]]:
        while True:
            with self._lock:
                now = time.monotonic()
                wait = self._wait_seconds(now, tokens)
                if wait <= 0:
                    if self.limits.requests_per_minute is not None:
                        self._requests.append(now)
                    if self.limits.tokens_per_minute is None:
                        return None
                    entry = [now, float(tokens)]
                    self._tokens.append(entry)
                    return entry
            await asyncio.sleep(wait)

    def _loop_state(self) -> _LoopState:
        loop = asyncio.get_running_loop()
        state = self._loops.get(loop)
        if state is None:
            state = self._loops[loop] = _LoopState()
        return state

    @asynccontextmanager
    async def slot(self, estimated_tokens: int = 0) -> AsyncIterator[_Slot]:
        """Wait for a concurrency slot and for room in the rate windows."""
        state = self._loop_state()
        async with state.condition:
            await state.condition.wait_for(
                lambda: (
                    self.limits.max_concurrency is None
                    or state.active < self.limits.max_concurrency
                )
            )
            state.active += 1
        try:
            entry = await self._reserve(estimated_tokens)
            yield _Slot(self, entry)
        finally:
            async with state.condition:
                state.active -= 1
                state.condition.notify_all()

    # --- calling ------------------------------------------------------------------

    def backoff(self, attempt: int) -> float:
        """Random wait before retry number `attempt` (0-based): full jitter."""
        cap = min(self.limits.max_backoff, self.limits.initial_backoff * 2**attempt)
        return random.uniform(0, cap)

    pipeline. Concrete subclasses must implement the `extract` method, which takes

    `__call__`, allowing extractor instances to be invoked directly on a list of
    strings.

                represented either as a list of `ExtractedConcept` objects or as a
                `ListOfExtractedConcepts` instance, depending on the implementation.

    class GeminiExtractor(BaseExtractor):
    """Extractor implementation backed by Gemini generative models.

    This extractor formats each input text into a prompt template, sends the prompt
    to a Gemini model configured to return JSON, validates the model response
    against the `ListOfExtractedConcepts` schema, and post-processes the extracted
    concepts.

    Post-processing includes:

    - computing character-level start and end offsets for each extracted concept
      based on its first occurrence in the original text;
    - discarding concepts not found in the original text;
    - optionally filtering concepts to a predefined set of allowed labels.

    This class is useful for zero-shot or few-shot structured extraction workflows
    where the model is expected to emit JSON matching the package's extraction
    schema.
    """

    def __init__(
        self,
        model_id: str,
        api_key: Optional[str] = None,
    ) -> None:
        """Initialize a Gemini-based extractor.

        Args:
            model_id: Identifier of the Gemini model to be used for extraction.
            api_key: Optional Google API key used to initialize the Gemini client.
                If omitted, authentication may rely on environment configuration.

        Raises:
            AssertionError: If the prompt template does not contain the `{text}`
                placeholder.
        """

        self.model_id = model_id
        self.api_key = api_key
        self.client = gemini_client(api_key)

    def _get_start_and_end_indices(self, text: str, prompt: str) -> List[int]:
        """Compute character offsets for an extracted span in the source text.

        This method finds the first occurrence of the extracted text in the original
        input text and returns its start and end indices.

        Args:
            text: The original source text.
            prompt: The extracted text span whose position should be located in the
                source text.

        Returns:
            A two-element list containing the start and end character offsets. If the
                span is not found, the start index will be `-1`, and the end index will be
                computed relative to that value.
        """
        start_index = text.find(prompt)
        end_index = start_index + len(prompt)
        return [start_index, end_index]

    def extract(self, tasks: List[ExtractionTask]) -> List[List[ExtractedConcept]]:
        """Extract structured concepts from a batch of texts using Gemini.

        For each input text, this method formats the configured prompt template,
        requests structured JSON output from the Gemini model, validates the response
        as `ListOfExtractedConcepts`, and post-processes the extracted concepts by
        assigning character offsets and applying optional label filtering.

        Args:
            tasks: A list of `ExtractionTask` instances.

        Returns:
            A list of `ListOfExtractedConcepts` objects, with one validated extraction
                result per input text.

        Notes:
            Character offsets are computed using the first occurrence of each extracted
                concept text in the original input text. If the same extracted span appears
                multiple times in the source text, this method does not disambiguate among
                occurrences.
        """

        results: List[List[ExtractedConcept]] = []
        for task in tasks:
            current_task_results: List[ExtractedConcept] = []
            assert isinstance(task, ExtractionTask), (
                "Input must be a list of ExtractionTask instances."
            )
            for idx, text in enumerate(task.texts):
                if task.prompt_args is None:
                    curr_prompt_args = {}
                elif isinstance(task.prompt_args, list):
                    curr_prompt_args = task.prompt_args[idx]
                elif isinstance(task.prompt_args, dict):
                    curr_prompt_args = task.prompt_args

                prompt = task.prompt_template.format(text=text, **curr_prompt_args)
                response = self.client.models.generate_content(
                    model=self.model_id,
                    contents=prompt,  # type: ignore[arg-type]
                    config={
                        "response_mime_type": "application/json",
                        "response_json_schema": task.data_model.model_json_schema(),
                    },
                )

                if response.text is None:
                    raise ValueError("No response returned from Gemini model.")

                result = task.data_model.model_validate_json(response.text)
                current_task_results.append(result)  # type: ignore[arg-type]
            results.append(current_task_results)

        return results
