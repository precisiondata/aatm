    Concept search agent

    Agentic concept search: an LLM agent that searches the vocabularies for the right code.

    One retrieval on a short span often misses the right concept. ConceptSearchAgent gets the text and its context and searches with tools (semantic search, lexical search, code lookup, concept details, hierarchy) until it finds the standard concept or concludes there is none. It is available in both pipelines:

    • Notes to OMOP tables: OmopTableBuilder(definitions, mapper=AgenticCoder(...)) (aatm.omop.AgenticCoder). The builder passes each text's related values and note line.
    • Terminology mapping: TerminologyMapper(selector=AgenticSelector(...)). The agent starts from the mapper's retrieved candidates, with each source concept's code and vocabulary as context.

    See aatm.concept_search.agent for the confidence scale of the scores.

    def cache_key(self) -> str:
        """Identical contexts share a key, so they are searched once."""
        return json.dumps(self.model_dump(), sort_keys=True, ensure_ascii=False)

    def describe(self) -> str:
        lines = [f"Text to code: {self.text!r}"]
        if self.related:
            lines.append(
                "Related values: "
                + "; ".join(f"{k}={v!r}" for k, v in self.related.items())
            )
        if self.sentence and self.sentence.strip() != self.text:
            lines.append(f"Sentence: {self.sentence.strip()!r}")
        if self.target:
            lines.append(f"Target: {self.target}")
        return "\n".join(lines)

    class ConceptSearchAgent:
    """Searches the vocabularies for the standard concept of a text, with tools.

    Args:
        model: An `aatm.llm` model id such as ``"openai/gpt-5-mini"``, or a
            `aatm.llm.StructuredLLM`.
        tools: The search tools.
        max_tool_calls: Tool calls the agent may make per text, after the initial
            semantic search. When they run out, the agent must answer with what it
            has seen, and its confidence is capped at ``low``.
        settings: Generation settings for a model given by id. For more
            reproducible runs, set ``temperature=0`` on models that accept it
            (reasoning models such as GPT-5 reject it); the trace makes runs
            comparable either way.
        top_k: Results per search.
    """

    def __init__(
        self,
        model: str | StructuredLLM,
        tools: ConceptSearchTools,
        max_tool_calls: int = 8,
        settings: Optional[LLMSettings | dict[str, Any]] = None,
        top_k: int = 10,
    ):
        if max_tool_calls < 0:
            raise ValueError("max_tool_calls must be >= 0")
        if isinstance(model, str):
            model = create_llm(model, settings=settings)
        self.llm: StructuredLLM = model
        self.tools = tools
        self.max_tool_calls = max_tool_calls
        self.top_k = top_k

    async def search(
        self,
        context: CodingContext,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        initial_candidates: Optional[Sequence[Candidate]] = None,
    ) -> ConceptSearchResult:
        """Find the standard concept of `context.text` within a domain and vocabularies.

        Args:
            context: The text and its context.
            domain: Only concepts of this domain qualify.
            vocabularies: Only concepts of these vocabularies qualify.
            initial_candidates: Candidates already retrieved for the text, e.g. by a
                `TerminologyMapper`. When ``None``, the agent starts with a semantic
                search on the text (or a lexical search without a retriever).

        Returns:
            The result. Errors from the model are caught and reported in
            ``error``, with no concept.
        """
        start = time.perf_counter()
        result = ConceptSearchResult()
        with track_usage() as usage:
            try:
                await self._run(
                    context, domain, vocabularies, initial_candidates, result
                )
            except Exception as e:  # noqa: BLE001 - one text must not stop a batch
                logger.warning("Concept search failed for %r: %s", context.text, e)
                result.concept = None
                result.confidence = None
                result.score = None
                result.error = f"{type(e).__name__}: {e}"
                result.rationale = f"Search failed: {result.error}"
        result.llm_calls = usage.calls
        result.input_tokens = usage.input_tokens
        result.output_tokens = usage.output_tokens
        result.seconds = time.perf_counter() - start
        return result

    async def _run(
        self,
        context: CodingContext,
        domain: Optional[str],
        vocabularies: Optional[Sequence[str]],
        initial_candidates: Optional[Sequence[Candidate]],
        result: ConceptSearchResult,
    ) -> None:
        seen: dict[int, Candidate] = {}
        history: list[str] = []

        def observe(candidates: Sequence[Candidate]) -> str:
            for c in candidates:
                if c.standard is not False and in_scope(c, domain, vocabularies):
                    seen.setdefault(c.concept_id, c)
            if not candidates:
                return "No results."
            return "\n".join(c.line() for c in candidates)

        # Step 0: the candidates of a plain retrieval, so easy texts need one LLM call.
        if initial_candidates is not None:
            first = list(initial_candidates)
            label = f"retrieved candidates for {context.text!r}"
            result.trace.append(
                {"action": "retrieved", "results": [c.short() for c in first]}
            )
        else:
            action = (
                "semantic_search"
                if self.tools.retriever is not None
                else "lexical_search"
            )
            first = await self._tool(action, context.text, domain, vocabularies)
            result.tool_calls += 1
            label = f"{action} {context.text!r}"
            result.trace.append(
                {
                    "action": action,
                    "query": context.text,
                    "results": [c.short() for c in first],
                }
            )
        history.append(f"[{label}]\n{observe(first)}")

        system = SYSTEM_PROMPT.format(
            scope=_scope_text(domain, vocabularies),
            actions="\n".join(ACTION_HELP[a] for a in self.tools.actions),
        )
        calls_left = self.max_tool_calls
        retries_left = 2  # invalid finishes or unavailable actions
        while True:
            forced = calls_left <= 0
            user = self._prompt(context, history, calls_left)
            step = await self.llm.generate(
                system=system, user=user, output_type=AgentStep
            )
            entry: dict[str, Any] = {"action": step.action, "reasoning": step.reasoning}

            if step.action == "finish" or forced:
                if step.action != "finish":
                    # The model ignored the instruction to finish: take nothing.
                    entry["error"] = "budget exhausted"
                    result.trace.append(entry)
                    result.budget_exhausted = True
                    result.rationale = "Tool budget exhausted before an answer."
                    return
                entry.update(concept_id=step.concept_id, confidence=step.confidence)
                problem = self._finish(step, seen, domain, vocabularies, result)
                if problem is None:
                    if forced:
                        result.budget_exhausted = True
                        if result.concept is not None:
                            result.confidence = "low"
                            result.score = CONFIDENCE_SCORES["low"]
                    result.trace.append(entry)
                    return
                entry["error"] = problem
                result.trace.append(entry)
                if forced or retries_left == 0:
                    result.concept = None
                    result.confidence = result.score = None
                    result.budget_exhausted = forced
                    result.rationale = f"No valid answer: {problem}"
                    return
                retries_left -= 1
                history.append(f"[finish {step.concept_id}]\nRejected: {problem}")
                continue

            if step.action not in self.tools.actions:
                problem = f"{step.action} is not available. Use one of {self.tools.actions} or finish."
            else:
                problem = self._missing_argument(step)
            if problem is not None:
                entry["error"] = problem
                result.trace.append(entry)
                if retries_left == 0:
                    result.rationale = f"No valid answer: {problem}"
                    return
                retries_left -= 1
                history.append(f"[{step.action}]\nError: {problem}")
                continue

            calls_left -= 1
            result.tool_calls += 1
            label, observation, summary = await self._execute(
                step, domain, vocabularies, observe
            )
            entry.update(summary)
            result.trace.append(entry)
            history.append(f"[{label}]\n{observation}")

    @staticmethod
    def _missing_argument(step: AgentStep) -> Optional[str]:
        if step.action in ("semantic_search", "lexical_search", "code_lookup"):
            if not (step.query or "").strip():
                return f"{step.action} needs a query."
        if step.action in ("concept_details", "hierarchy") and step.concept_id is None:
            return f"{step.action} needs a concept_id."
        if step.action == "hierarchy" and step.direction is None:
            return "hierarchy needs a direction: parents or children."
        return None

    async def _tool(
        self,
        action: str,
        query: str,
        domain: Optional[str],
        vocabularies: Optional[Sequence[str]],
    ) -> list[Candidate]:
        method = getattr(self.tools, action)
        # The tools block (SQLite, the vector database): run them off the event loop.
        return await asyncio.to_thread(method, query, domain, vocabularies, self.top_k)

    async def _execute(
        self,
        step: AgentStep,
        domain: Optional[str],
        vocabularies: Optional[Sequence[str]],
        observe: Any,
    ) -> tuple[str, str, dict[str, Any]]:
        if step.action in ("semantic_search", "lexical_search"):
            query = step.query.strip()  # type: ignore[union-attr]
            found = await self._tool(step.action, query, domain, vocabularies)
            return (
                f"{step.action} {query!r}",
                observe(found),
                {"query": query, "results": [c.short() for c in found]},
            )
        if step.action == "code_lookup":
            code = step.query.strip()  # type: ignore[union-attr]
            found = await asyncio.to_thread(
                self.tools.code_lookup, code, domain, vocabularies
            )
            return (
                f"code_lookup {code!r}",
                observe(found),
                {"query": code, "results": [c.short() for c in found]},
            )
        if step.action == "concept_details":
            assert step.concept_id is not None  # checked by _missing_argument
            details = await asyncio.to_thread(
                self.tools.concept_details, step.concept_id
            )
            return (
                f"concept_details {step.concept_id}",
                details,
                {"concept_id": step.concept_id},
            )
        assert step.action == "hierarchy"
        # Checked by _missing_argument.
        assert step.concept_id is not None and step.direction is not None
        found = await asyncio.to_thread(
            self.tools.hierarchy,
            step.concept_id,
            step.direction,
            domain,
            vocabularies,
            self.top_k,
        )
        return (
            f"hierarchy {step.direction} of {step.concept_id}",
            observe(found),
            {
                "concept_id": step.concept_id,
                "direction": step.direction,
                "results": [c.short() for c in found],
            },
        )

    def _finish(
        self,
        step: AgentStep,
        seen: dict[int, Candidate],
        domain: Optional[str],
        vocabularies: Optional[Sequence[str]],
        result: ConceptSearchResult,
    ) -> Optional[str]:
        """Record a finish step; return why it is rejected, or ``None``."""
        result.rationale = step.reasoning
        if step.concept_id is None:
            result.concept = None
            result.confidence = result.score = None
            return None
        if step.confidence is None:
            return "finish with a concept needs a confidence: high, medium or low."
        candidate = seen.get(step.concept_id)
        if candidate is None:
            # Not returned in scope by a tool: accept it only if the vocabulary
            # confirms it qualifies.
            vocabulary = self.tools.vocabulary
            concept = (
                vocabulary.by_id([step.concept_id]).get(step.concept_id)
                if vocabulary is not None
                else None
            )
            if concept is None or not in_scope(concept, domain, vocabularies):
                return (
                    f"{step.concept_id} is not a standard concept "
                    f"{_scope_text(domain, vocabularies)} returned by a tool."
                )
            candidate = Candidate.from_concept(concept)
        result.concept = candidate
        result.confidence = step.confidence
        result.score = CONFIDENCE_SCORES[step.confidence]
        return None

    @staticmethod
    def _prompt(context: CodingContext, history: list[str], calls_left: int) -> str:
        budget = (
            f"Tool calls left: {calls_left}."
            if calls_left > 0
            else "No tool calls left: you must finish now, with a concept from the "
            "results above or null."
        )
        return (
            f"{context.describe()}\n\n"
            "Results so far:\n\n"
            + "\n\n".join(history)
            + f"\n\n{budget} Choose the next action."
        )

    async def search_many(
        self,
        items: Sequence[tuple[CodingContext, Optional[Sequence[Candidate]]]],
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        max_concurrency: int = 8,
    ) -> list[ConceptSearchResult]:
        """Search several texts concurrently; identical contexts are searched once.

        Args:
            items: ``(context, initial_candidates)`` pairs.
            domain: See `search`.
            vocabularies: See `search`.
            max_concurrency: Searches running at the same time.

        Returns:
            One result per item, in order.
        """
        semaphore = asyncio.Semaphore(max_concurrency)
        unique: dict[str, tuple[CodingContext, Optional[Sequence[Candidate]]]] = {}
        for context, initial in items:
            unique.setdefault(context.cache_key(), (context, initial))

        async def run(context: CodingContext, initial: Any) -> ConceptSearchResult:
            async with semaphore:
                return await self.search(context, domain, vocabularies, initial)

        keys = list(unique)
        results = await asyncio.gather(*(run(*unique[k]) for k in keys))
        by_key = dict(zip(keys, results))
        return [by_key[context.cache_key()] for context, _ in items]

    self,
    context: CodingContext,
    domain: Optional[str] = None,
    vocabularies: Optional[Sequence[str]] = None,
    initial_candidates: Optional[Sequence[Candidate]] = None,
) -> ConceptSearchResult:
    """Find the standard concept of `context.text` within a domain and vocabularies.

    Args:
        context: The text and its context.
        domain: Only concepts of this domain qualify.
        vocabularies: Only concepts of these vocabularies qualify.

            search on the text (or a lexical search without a retriever).

    Returns:
        The result. Errors from the model are caught and reported in
        ``error``, with no concept.
    """
    start = time.perf_counter()
    result = ConceptSearchResult()
    with track_usage() as usage:
        try:
            await self._run(
                context, domain, vocabularies, initial_candidates, result
            )
        except Exception as e:  # noqa: BLE001 - one text must not stop a batch
            logger.warning("Concept search failed for %r: %s", context.text, e)

            result.confidence = None
            result.score = None
            result.error = f"{type(e).__name__}: {e}"
            result.rationale = f"Search failed: {result.error}"
    result.llm_calls = usage.calls
    result.input_tokens = usage.input_tokens
    result.output_tokens = usage.output_tokens
    result.seconds = time.perf_counter() - start
    return result

        One result per item, in order.

            return await self.search(context, domain, vocabularies, initial)

    results = await asyncio.gather(*(run(*unique[k]) for k in keys))
    by_key = dict(zip(keys, results))
    return [by_key[context.cache_key()] for context, _ in items]

    class ConceptSearchTools:
    """The agent's tools over a retriever and a vocabulary database.

    Args:
        retriever: A retriever with a ``retrieve(queries, where=..., top_k=...)``
            method, such as `aatm.retrievers.ChromaDBRetriever` or
            `aatm.retrievers.PineconeRetriever`, for semantic search.
        vocabulary: The vocabulary database, as a path or a `VocabularySearch`, for
            lexical search, code lookup, concept details and hierarchy. ``None``
            leaves only semantic search.
        top_k: Number of results per search.
    """

    def __init__(
        self,
        retriever: Any = None,
        vocabulary: VocabularySearch | str | Path | None = None,
        top_k: int = 10,
    ):
        if retriever is None and vocabulary is None:
            raise ValueError(
                "The concept search needs a retriever, a vocabulary, or both"
            )
        self.retriever = retriever
        self.vocabulary = (
            VocabularySearch(vocabulary)
            if isinstance(vocabulary, (str, Path))
            else vocabulary
        )
        self.top_k = top_k

    @property
    def actions(self) -> list[str]:
        """The tools available with this configuration."""
        actions = []
        if self.retriever is not None:
            actions.append("semantic_search")
        if self.vocabulary is not None:
            actions += ["lexical_search", "code_lookup", "concept_details", "hierarchy"]
        return actions

    def semantic_search(
        self,
        query: str,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        top_k: Optional[int] = None,
    ) -> list[Candidate]:
        if self.retriever is None:
            raise ValueError("semantic_search needs a retriever")
        k = top_k or self.top_k
        # Several indexed expressions (synonyms) can point to the same concept:
        # retrieve more to return about k distinct concepts.
        results = self.retriever.retrieve(
            [query], where=retriever_filter(domain, vocabularies), top_k=k * 2
        )
        return candidates_from_expressions(results.results[0])[:k]

    def lexical_search(
        self,
        query: str,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        top_k: Optional[int] = None,
    ) -> list[Candidate]:
        assert self.vocabulary is not None
        return self.vocabulary.search_names(
            query, domain, vocabularies, limit=top_k or self.top_k
        )

    def code_lookup(
        self,
        code: str,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
    ) -> list[Candidate]:
        """The concepts with this code, and the in-scope standard concepts they map to."""
        assert self.vocabulary is not None
        found = self.vocabulary.search_code(code)
        maps_to = self.vocabulary.maps_to(c.concept_id for c in found if not c.standard)
        targets = self.vocabulary.by_id({t for ts in maps_to.values() for t in ts})
        results: list[Candidate] = []
        for candidate in found:
            results.append(candidate)
            for target_id in maps_to.get(candidate.concept_id, []):
                target = targets.get(target_id)
                if target is not None:
                    mapped = Candidate.from_concept(target)
                    mapped.matched = (
                        f"maps from {candidate.vocabulary_id} {candidate.concept_code}"
                    )
                    results.append(mapped)
        return [
            c for c in results if not c.standard or in_scope(c, domain, vocabularies)
        ]

    def concept_details(self, concept_id: int) -> str:
        assert self.vocabulary is not None
        concept = self.vocabulary.by_id([concept_id]).get(concept_id)
        if concept is None:
            return f"No concept with id {concept_id}."
        lines = [
            f"concept_id: {concept.concept_id}",
            f"name: {concept.concept_name}",
            f"vocabulary: {concept.vocabulary_id} {concept.concept_code}",
            f"domain: {concept.domain_id}",
            f"standard: {'yes' if concept.is_standard else 'no'}",
        ]
        if concept.invalid_reason:
            lines.append(f"invalid_reason: {concept.invalid_reason}")
        synonyms = self.vocabulary.synonyms(concept_id)
        if synonyms:
            lines.append(f"synonyms: {'; '.join(synonyms)}")
        relationships = self.vocabulary.relationships(concept_id)
        related_ids = {
            target
            for rel, target in relationships
            if rel in LOINC_AXES or rel == "Maps to"
        }
        names = self.vocabulary.by_id(related_ids)
        axes = {
            LOINC_AXES[rel]: names[target].concept_name
            for rel, target in relationships
            if rel in LOINC_AXES and target in names
        }
        if axes:
            lines.append(
                "LOINC axes: "
                + "; ".join(
                    f"{axis}={axes[axis]}"
                    for axis in LOINC_AXES.values()
                    if axis in axes
                )
            )
        maps_to = [
            names[target]
            for rel, target in relationships
            if rel == "Maps to" and target in names and target != concept_id
        ]
        for target in maps_to:
            lines.append(f"maps to: {Candidate.from_concept(target).line()}")
        return "\n".join(lines)

    def hierarchy(
        self,
        concept_id: int,
        direction: Literal["parents", "children"],
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        top_k: Optional[int] = None,

        assert self.vocabulary is not None
        return self.vocabulary.neighbours(
            concept_id, direction, domain, vocabularies, limit=(top_k or self.top_k) * 2
        )

    class VocabularySearch(SqliteConceptLookup):
    """Searches in the SQLite vocabulary database built by ``aatm init``.

    The tables are not indexed, so each search scans a table once (well under a
    second for the concept table of a typical download).
    """

    def __init__(self, path: str | Path = DEFAULT_VOCABULARY_DB):
        super().__init__(path)

    @staticmethod
    def _scope_sql(
        domain: Optional[str], vocabularies: Optional[Sequence[str]], alias: str = "c"
    ) -> tuple[str, list]:
        sql = f" AND {alias}.standard_concept = 'S' AND {alias}.invalid_reason IS NULL"
        params: list = []
        if domain is not None:
            sql += f" AND {alias}.domain_id = ?"
            params.append(domain)
        if vocabularies:
            sql += (
                f" AND {alias}.vocabulary_id IN ({','.join('?' * len(vocabularies))})"
            )
            params.extend(vocabularies)
        return sql, params

    def search_names(
        self,
        query: str,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        limit: int = 20,
    ) -> list[Candidate]:
        """Standard concepts whose name or a synonym contains every word of `query`.

        Ranked by: exact name, name starting with the query, then shorter names, which
        are usually the more general concepts.
        """
        words = [w for w in query.casefold().split() if w]
        if not words:
            return []
        scope, scope_params = self._scope_sql(domain, vocabularies)
        like = " AND ".join("lower({col}) LIKE ?" for _ in words)
        like_params = [f"%{w}%" for w in words]
        rows = self._query(
            "SELECT c.concept_id, c.concept_name, c.domain_id, c.vocabulary_id, "
            "c.concept_code, c.standard_concept, c.invalid_reason, c.concept_name "
            f"FROM concept c WHERE {like.format(col='c.concept_name')}{scope} "
            "UNION "
            "SELECT c.concept_id, c.concept_name, c.domain_id, c.vocabulary_id, "
            "c.concept_code, c.standard_concept, c.invalid_reason, s.concept_synonym_name "
            "FROM concept_synonym s JOIN concept c ON c.concept_id = s.concept_id "
            f"WHERE {like.format(col='s.concept_synonym_name')}{scope}",
            [*like_params, *scope_params, *like_params, *scope_params],
        )
        q = query.casefold().strip()
        best: dict[int, tuple[tuple, Candidate]] = {}
        for row in rows:
            candidate = Candidate.from_concept(self._concept(row[:7]))
            matched = row[7] or candidate.concept_name
            candidate.matched = matched
            m = matched.casefold()
            rank = (0 if m == q else 1 if m.startswith(q) else 2, len(m), m)
            current = best.get(candidate.concept_id)
            if current is None or rank < current[0]:
                best[candidate.concept_id] = (rank, candidate)
        ordered = sorted(best.values(), key=lambda item: item[0])
        return [candidate for _, candidate in ordered[:limit]]

    def search_code(self, code: str) -> list[Candidate]:

    self,
    query: str,
    domain: Optional[str] = None,
    vocabularies: Optional[Sequence[str]] = None,
    limit: int = 20,

    """Standard concepts whose name or a synonym contains every word of `query`.

    Ranked by: exact name, name starting with the query, then shorter names, which
    are usually the more general concepts.
    """
    words = [w for w in query.casefold().split() if w]
    if not words:
        return []
    scope, scope_params = self._scope_sql(domain, vocabularies)
    like = " AND ".join("lower({col}) LIKE ?" for _ in words)
    like_params = [f"%{w}%" for w in words]
    rows = self._query(
        "SELECT c.concept_id, c.concept_name, c.domain_id, c.vocabulary_id, "
        "c.concept_code, c.standard_concept, c.invalid_reason, c.concept_name "
        f"FROM concept c WHERE {like.format(col='c.concept_name')}{scope} "
        "UNION "
        "SELECT c.concept_id, c.concept_name, c.domain_id, c.vocabulary_id, "
        "c.concept_code, c.standard_concept, c.invalid_reason, s.concept_synonym_name "
        "FROM concept_synonym s JOIN concept c ON c.concept_id = s.concept_id "
        f"WHERE {like.format(col='s.concept_synonym_name')}{scope}",
        [*like_params, *scope_params, *like_params, *scope_params],
    )
    q = query.casefold().strip()
    best: dict[int, tuple[tuple, Candidate]] = {}
    for row in rows:
        candidate = Candidate.from_concept(self._concept(row[:7]))
        matched = row[7] or candidate.concept_name
        candidate.matched = matched
        m = matched.casefold()
        rank = (0 if m == q else 1 if m.startswith(q) else 2, len(m), m)
        current = best.get(candidate.concept_id)
        if current is None or rank < current[0]:
            best[candidate.concept_id] = (rank, candidate)
    ordered = sorted(best.values(), key=lambda item: item[0])
    return [candidate for _, candidate in ordered[:limit]]

    A single retrieval on a short span often misses the right concept: "creatinine" says nothing of the specimen, and "Hb" or "potássio" need expanding or translating before an embedding search finds them. )

    The agent is used by aatm.omop.AgenticCoder (extraction to OMOP tables) and by aatm.concept_search.AgenticSelector (terminology mapping with TerminologyMapper).

    The loop is provider-agnostic: at each step the model answers with one structured AgentStep (an action and its arguments), so any aatm.llm model works, without native tool calling.

    Confidence scale

    The agent reports a confidence level, stored as a score:

    Level Score Meaning
    high 0.9 The concept matches the text and everything the context states (e.g. specimen and unit).
    medium 0.6 The right concept, but a detail the context does not state was assumed (e.g. serum for a chemistry test without specimen).
    low 0.3 The closest concept available, with a known mismatch or real ambiguity.

    When the tool budget runs out, the agent must answer with what it has seen, and the score is capped at low. min_score: 0.5 therefore keeps high and medium.

    class ConceptSearchAgent:
    """Searches the vocabularies for the standard concept of a text, with tools.

    Args:
        model: An `aatm.llm` model id such as ``"openai/gpt-5-mini"``, or a
            `aatm.llm.StructuredLLM`.
        tools: The search tools.
        max_tool_calls: Tool calls the agent may make per text, after the initial
            semantic search. When they run out, the agent must answer with what it
            has seen, and its confidence is capped at ``low``.
        settings: Generation settings for a model given by id. For more
            reproducible runs, set ``temperature=0`` on models that accept it
            (reasoning models such as GPT-5 reject it); the trace makes runs
            comparable either way.
        top_k: Results per search.
    """

    def __init__(
        self,
        model: str | StructuredLLM,
        tools: ConceptSearchTools,
        max_tool_calls: int = 8,
        settings: Optional[LLMSettings | dict[str, Any]] = None,
        top_k: int = 10,
    ):
        if max_tool_calls < 0:
            raise ValueError("max_tool_calls must be >= 0")
        if isinstance(model, str):
            model = create_llm(model, settings=settings)
        self.llm: StructuredLLM = model
        self.tools = tools
        self.max_tool_calls = max_tool_calls
        self.top_k = top_k

    async def search(
        self,
        context: CodingContext,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        initial_candidates: Optional[Sequence[Candidate]] = None,
    ) -> ConceptSearchResult:
        """Find the standard concept of `context.text` within a domain and vocabularies.

        Args:
            context: The text and its context.
            domain: Only concepts of this domain qualify.
            vocabularies: Only concepts of these vocabularies qualify.
            initial_candidates: Candidates already retrieved for the text, e.g. by a
                `TerminologyMapper`. When ``None``, the agent starts with a semantic
                search on the text (or a lexical search without a retriever).

        Returns:
            The result. Errors from the model are caught and reported in
            ``error``, with no concept.
        """
        start = time.perf_counter()
        result = ConceptSearchResult()
        with track_usage() as usage:
            try:
                await self._run(
                    context, domain, vocabularies, initial_candidates, result
                )
            except Exception as e:  # noqa: BLE001 - one text must not stop a batch
                logger.warning("Concept search failed for %r: %s", context.text, e)
                result.concept = None
                result.confidence = None
                result.score = None
                result.error = f"{type(e).__name__}: {e}"
                result.rationale = f"Search failed: {result.error}"
        result.llm_calls = usage.calls
        result.input_tokens = usage.input_tokens
        result.output_tokens = usage.output_tokens
        result.seconds = time.perf_counter() - start
        return result

    async def _run(
        self,
        context: CodingContext,
        domain: Optional[str],
        vocabularies: Optional[Sequence[str]],
        initial_candidates: Optional[Sequence[Candidate]],
        result: ConceptSearchResult,
    ) -> None:
        seen: dict[int, Candidate] = {}
        history: list[str] = []

        def observe(candidates: Sequence[Candidate]) -> str:
            for c in candidates:
                if c.standard is not False and in_scope(c, domain, vocabularies):
                    seen.setdefault(c.concept_id, c)
            if not candidates:
                return "No results."
            return "\n".join(c.line() for c in candidates)

        # Step 0: the candidates of a plain retrieval, so easy texts need one LLM call.
        if initial_candidates is not None:
            first = list(initial_candidates)
            label = f"retrieved candidates for {context.text!r}"
            result.trace.append(
                {"action": "retrieved", "results": [c.short() for c in first]}
            )
        else:
            action = (
                "semantic_search"
                if self.tools.retriever is not None
                else "lexical_search"
            )
            first = await self._tool(action, context.text, domain, vocabularies)
            result.tool_calls += 1
            label = f"{action} {context.text!r}"
            result.trace.append(
                {
                    "action": action,
                    "query": context.text,
                    "results": [c.short() for c in first],
                }
            )
        history.append(f"[{label}]\n{observe(first)}")

        system = SYSTEM_PROMPT.format(
            scope=_scope_text(domain, vocabularies),
            actions="\n".join(ACTION_HELP[a] for a in self.tools.actions),
        )
        calls_left = self.max_tool_calls
        retries_left = 2  # invalid finishes or unavailable actions
        while True:
            forced = calls_left <= 0
            user = self._prompt(context, history, calls_left)
            step = await self.llm.generate(
                system=system, user=user, output_type=AgentStep
            )
            entry: dict[str, Any] = {"action": step.action, "reasoning": step.reasoning}

            if step.action == "finish" or forced:
                if step.action != "finish":
                    # The model ignored the instruction to finish: take nothing.
                    entry["error"] = "budget exhausted"
                    result.trace.append(entry)
                    result.budget_exhausted = True
                    result.rationale = "Tool budget exhausted before an answer."
                    return
                entry.update(concept_id=step.concept_id, confidence=step.confidence)
                problem = self._finish(step, seen, domain, vocabularies, result)
                if problem is None:
                    if forced:
                        result.budget_exhausted = True
                        if result.concept is not None:
                            result.confidence = "low"
                            result.score = CONFIDENCE_SCORES["low"]
                    result.trace.append(entry)
                    return
                entry["error"] = problem
                result.trace.append(entry)
                if forced or retries_left == 0:
                    result.concept = None
                    result.confidence = result.score = None
                    result.budget_exhausted = forced
                    result.rationale = f"No valid answer: {problem}"
                    return
                retries_left -= 1
                history.append(f"[finish {step.concept_id}]\nRejected: {problem}")
                continue

            if step.action not in self.tools.actions:
                problem = f"{step.action} is not available. Use one of {self.tools.actions} or finish."
            else:
                problem = self._missing_argument(step)
            if problem is not None:
                entry["error"] = problem
                result.trace.append(entry)
                if retries_left == 0:
                    result.rationale = f"No valid answer: {problem}"
                    return
                retries_left -= 1
                history.append(f"[{step.action}]\nError: {problem}")
                continue

            calls_left -= 1
            result.tool_calls += 1
            label, observation, summary = await self._execute(
                step, domain, vocabularies, observe
            )
            entry.update(summary)
            result.trace.append(entry)
            history.append(f"[{label}]\n{observation}")

    @staticmethod
    def _missing_argument(step: AgentStep) -> Optional[str]:
        if step.action in ("semantic_search", "lexical_search", "code_lookup"):
            if not (step.query or "").strip():
                return f"{step.action} needs a query."
        if step.action in ("concept_details", "hierarchy") and step.concept_id is None:
            return f"{step.action} needs a concept_id."
        if step.action == "hierarchy" and step.direction is None:
            return "hierarchy needs a direction: parents or children."
        return None

    async def _tool(
        self,
        action: str,
        query: str,
        domain: Optional[str],
        vocabularies: Optional[Sequence[str]],
    ) -> list[Candidate]:
        method = getattr(self.tools, action)
        # The tools block (SQLite, the vector database): run them off the event loop.
        return await asyncio.to_thread(method, query, domain, vocabularies, self.top_k)

    async def _execute(
        self,
        step: AgentStep,
        domain: Optional[str],
        vocabularies: Optional[Sequence[str]],
        observe: Any,
    ) -> tuple[str, str, dict[str, Any]]:
        if step.action in ("semantic_search", "lexical_search"):
            query = step.query.strip()  # type: ignore[union-attr]
            found = await self._tool(step.action, query, domain, vocabularies)
            return (
                f"{step.action} {query!r}",
                observe(found),
                {"query": query, "results": [c.short() for c in found]},
            )
        if step.action == "code_lookup":
            code = step.query.strip()  # type: ignore[union-attr]
            found = await asyncio.to_thread(
                self.tools.code_lookup, code, domain, vocabularies
            )
            return (
                f"code_lookup {code!r}",
                observe(found),
                {"query": code, "results": [c.short() for c in found]},
            )
        if step.action == "concept_details":
            assert step.concept_id is not None  # checked by _missing_argument
            details = await asyncio.to_thread(
                self.tools.concept_details, step.concept_id
            )
            return (
                f"concept_details {step.concept_id}",
                details,
                {"concept_id": step.concept_id},
            )
        assert step.action == "hierarchy"
        # Checked by _missing_argument.
        assert step.concept_id is not None and step.direction is not None
        found = await asyncio.to_thread(
            self.tools.hierarchy,
            step.concept_id,
            step.direction,
            domain,
            vocabularies,
            self.top_k,
        )
        return (
            f"hierarchy {step.direction} of {step.concept_id}",
            observe(found),
            {
                "concept_id": step.concept_id,
                "direction": step.direction,
                "results": [c.short() for c in found],
            },
        )

    def _finish(
        self,
        step: AgentStep,
        seen: dict[int, Candidate],
        domain: Optional[str],
        vocabularies: Optional[Sequence[str]],
        result: ConceptSearchResult,
    ) -> Optional[str]:
        """Record a finish step; return why it is rejected, or ``None``."""
        result.rationale = step.reasoning
        if step.concept_id is None:
            result.concept = None
            result.confidence = result.score = None
            return None
        if step.confidence is None:
            return "finish with a concept needs a confidence: high, medium or low."
        candidate = seen.get(step.concept_id)
        if candidate is None:
            # Not returned in scope by a tool: accept it only if the vocabulary
            # confirms it qualifies.
            vocabulary = self.tools.vocabulary
            concept = (
                vocabulary.by_id([step.concept_id]).get(step.concept_id)
                if vocabulary is not None
                else None
            )
            if concept is None or not in_scope(concept, domain, vocabularies):
                return (
                    f"{step.concept_id} is not a standard concept "
                    f"{_scope_text(domain, vocabularies)} returned by a tool."
                )
            candidate = Candidate.from_concept(concept)
        result.concept = candidate
        result.confidence = step.confidence
        result.score = CONFIDENCE_SCORES[step.confidence]
        return None

    @staticmethod
    def _prompt(context: CodingContext, history: list[str], calls_left: int) -> str:
        budget = (
            f"Tool calls left: {calls_left}."
            if calls_left > 0
            else "No tool calls left: you must finish now, with a concept from the "
            "results above or null."
        )
        return (
            f"{context.describe()}\n\n"
            "Results so far:\n\n"
            + "\n\n".join(history)
            + f"\n\n{budget} Choose the next action."
        )

    class VocabularySearch(SqliteConceptLookup):
    """Searches in the SQLite vocabulary database built by ``aatm init``.

    The tables are not indexed, so each search scans a table once (well under a
    second for the concept table of a typical download).
    """

    def __init__(self, path: str | Path = DEFAULT_VOCABULARY_DB):
        super().__init__(path)

    @staticmethod
    def _scope_sql(
        domain: Optional[str], vocabularies: Optional[Sequence[str]], alias: str = "c"
    ) -> tuple[str, list]:
        sql = f" AND {alias}.standard_concept = 'S' AND {alias}.invalid_reason IS NULL"
        params: list = []
        if domain is not None:
            sql += f" AND {alias}.domain_id = ?"
            params.append(domain)
        if vocabularies:
            sql += (
                f" AND {alias}.vocabulary_id IN ({','.join('?' * len(vocabularies))})"
            )
            params.extend(vocabularies)
        return sql, params

    def search_names(
        self,
        query: str,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        limit: int = 20,
    ) -> list[Candidate]:
        """Standard concepts whose name or a synonym contains every word of `query`.

        Ranked by: exact name, name starting with the query, then shorter names, which
        are usually the more general concepts.
        """
        words = [w for w in query.casefold().split() if w]
        if not words:
            return []
        scope, scope_params = self._scope_sql(domain, vocabularies)
        like = " AND ".join("lower({col}) LIKE ?" for _ in words)
        like_params = [f"%{w}%" for w in words]
        rows = self._query(
            "SELECT c.concept_id, c.concept_name, c.domain_id, c.vocabulary_id, "
            "c.concept_code, c.standard_concept, c.invalid_reason, c.concept_name "
            f"FROM concept c WHERE {like.format(col='c.concept_name')}{scope} "
            "UNION "
            "SELECT c.concept_id, c.concept_name, c.domain_id, c.vocabulary_id, "
            "c.concept_code, c.standard_concept, c.invalid_reason, s.concept_synonym_name "
            "FROM concept_synonym s JOIN concept c ON c.concept_id = s.concept_id "
            f"WHERE {like.format(col='s.concept_synonym_name')}{scope}",
            [*like_params, *scope_params, *like_params, *scope_params],
        )
        q = query.casefold().strip()
        best: dict[int, tuple[tuple, Candidate]] = {}
        for row in rows:
            candidate = Candidate.from_concept(self._concept(row[:7]))
            matched = row[7] or candidate.concept_name
            candidate.matched = matched
            m = matched.casefold()
            rank = (0 if m == q else 1 if m.startswith(q) else 2, len(m), m)
            current = best.get(candidate.concept_id)
            if current is None or rank < current[0]:
                best[candidate.concept_id] = (rank, candidate)
        ordered = sorted(best.values(), key=lambda item: item[0])
        return [candidate for _, candidate in ordered[:limit]]

    def search_code(self, code: str) -> list[Candidate]:
        """Concepts of any vocabulary whose code is `code`, ignoring case."""
        code = code.strip()
        if not code:
            return []
        rows = self._query(
            "SELECT concept_id, concept_name, domain_id, vocabulary_id, concept_code, "
            "standard_concept, invalid_reason FROM concept "
            "WHERE lower(concept_code) = ?",
            [code.lower()],
        )
        return [Candidate.from_concept(self._concept(row)) for row in rows]

    def relationships(self, concept_id: int) -> list[tuple[str, int]]:
        """Valid ``(relationship_id, concept_id_2)`` pairs of a concept."""
        rows = self._query(
            "SELECT relationship_id, concept_id_2 FROM concept_relationship "
            "WHERE concept_id_1 = ? AND invalid_reason IS NULL",
            [str(concept_id)],
        )
        return [(rel, int(target)) for rel, target in rows]

    def synonyms(self, concept_id: int, limit: int = 10) -> list[str]:
        rows = self._query(
            "SELECT concept_synonym_name FROM concept_synonym WHERE concept_id = ?",
            [str(concept_id)],
        )
        return sorted({r[0] for r in rows if r[0]})[:limit]

    def neighbours(
        self,
        concept_id: int,
        direction: Literal["parents", "children"],
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        limit: int = 25,
    ) -> list[Candidate]:
        """Direct parents or children of a standard concept (``concept_ancestor``)."""
        this, other = (
            ("descendant", "ancestor")
            if direction == "parents"
            else ("ancestor", "descendant")
        )
        scope, scope_params = self._scope_sql(domain, vocabularies)
        rows = self._query(

    def search_names(
    self,
    query: str,
    domain: Optional[str] = None,
    vocabularies: Optional[Sequence[str]] = None,
    limit: int = 20,
) -> list[Candidate]:
    """Standard concepts whose name or a synonym contains every word of `query`.

    Ranked by: exact name, name starting with the query, then shorter names, which
    are usually the more general concepts.
    """
    words = [w for w in query.casefold().split() if w]
    if not words:
        return []
    scope, scope_params = self._scope_sql(domain, vocabularies)
    like = " AND ".join("lower({col}) LIKE ?" for _ in words)
    like_params = [f"%{w}%" for w in words]
    rows = self._query(
        "SELECT c.concept_id, c.concept_name, c.domain_id, c.vocabulary_id, "
        "c.concept_code, c.standard_concept, c.invalid_reason, c.concept_name "
        f"FROM concept c WHERE {like.format(col='c.concept_name')}{scope} "
        "UNION "
        "SELECT c.concept_id, c.concept_name, c.domain_id, c.vocabulary_id, "
        "c.concept_code, c.standard_concept, c.invalid_reason, s.concept_synonym_name "
        "FROM concept_synonym s JOIN concept c ON c.concept_id = s.concept_id "
        f"WHERE {like.format(col='s.concept_synonym_name')}{scope}",
        [*like_params, *scope_params, *like_params, *scope_params],
    )
    q = query.casefold().strip()
    best: dict[int, tuple[tuple, Candidate]] = {}
    for row in rows:
        candidate = Candidate.from_concept(self._concept(row[:7]))
        matched = row[7] or candidate.concept_name
        candidate.matched = matched
        m = matched.casefold()
        rank = (0 if m == q else 1 if m.startswith(q) else 2, len(m), m)
        current = best.get(candidate.concept_id)
        if current is None or rank < current[0]:
            best[candidate.concept_id] = (rank, candidate)
    ordered = sorted(best.values(), key=lambda item: item[0])
    return [candidate for _, candidate in ordered[:limit]]

    class ConceptSearchTools:
    """The agent's tools over a retriever and a vocabulary database.

    Args:
        retriever: A retriever with a ``retrieve(queries, where=..., top_k=...)``
            method, such as `aatm.retrievers.ChromaDBRetriever` or
            `aatm.retrievers.PineconeRetriever`, for semantic search.
        vocabulary: The vocabulary database, as a path or a `VocabularySearch`, for
            lexical search, code lookup, concept details and hierarchy. ``None``
            leaves only semantic search.
        top_k: Number of results per search.
    """

    def __init__(
        self,
        retriever: Any = None,
        vocabulary: VocabularySearch | str | Path | None = None,
        top_k: int = 10,
    ):
        if retriever is None and vocabulary is None:
            raise ValueError(
                "The concept search needs a retriever, a vocabulary, or both"
            )
        self.retriever = retriever
        self.vocabulary = (
            VocabularySearch(vocabulary)
            if isinstance(vocabulary, (str, Path))
            else vocabulary
        )
        self.top_k = top_k

    @property
    def actions(self) -> list[str]:
        """The tools available with this configuration."""
        actions = []
        if self.retriever is not None:
            actions.append("semantic_search")
        if self.vocabulary is not None:
            actions += ["lexical_search", "code_lookup", "concept_details", "hierarchy"]
        return actions

    def semantic_search(
        self,
        query: str,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        top_k: Optional[int] = None,
    ) -> list[Candidate]:
        if self.retriever is None:
            raise ValueError("semantic_search needs a retriever")
        k = top_k or self.top_k
        # Several indexed expressions (synonyms) can point to the same concept:
        # retrieve more to return about k distinct concepts.
        results = self.retriever.retrieve(
            [query], where=retriever_filter(domain, vocabularies), top_k=k * 2
        )
        return candidates_from_expressions(results.results[0])[:k]

    def lexical_search(
        self,
        query: str,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        top_k: Optional[int] = None,
    ) -> list[Candidate]:
        assert self.vocabulary is not None
        return self.vocabulary.search_names(
            query, domain, vocabularies, limit=top_k or self.top_k
        )

    def code_lookup(
        self,
        code: str,
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
    ) -> list[Candidate]:
        """The concepts with this code, and the in-scope standard concepts they map to."""
        assert self.vocabulary is not None
        found = self.vocabulary.search_code(code)
        maps_to = self.vocabulary.maps_to(c.concept_id for c in found if not c.standard)
        targets = self.vocabulary.by_id({t for ts in maps_to.values() for t in ts})
        results: list[Candidate] = []
        for candidate in found:
            results.append(candidate)
            for target_id in maps_to.get(candidate.concept_id, []):
                target = targets.get(target_id)
                if target is not None:
                    mapped = Candidate.from_concept(target)
                    mapped.matched = (
                        f"maps from {candidate.vocabulary_id} {candidate.concept_code}"
                    )
                    results.append(mapped)
        return [
            c for c in results if not c.standard or in_scope(c, domain, vocabularies)
        ]

    def concept_details(self, concept_id: int) -> str:
        assert self.vocabulary is not None
        concept = self.vocabulary.by_id([concept_id]).get(concept_id)
        if concept is None:
            return f"No concept with id {concept_id}."
        lines = [
            f"concept_id: {concept.concept_id}",
            f"name: {concept.concept_name}",
            f"vocabulary: {concept.vocabulary_id} {concept.concept_code}",
            f"domain: {concept.domain_id}",
            f"standard: {'yes' if concept.is_standard else 'no'}",
        ]
        if concept.invalid_reason:
            lines.append(f"invalid_reason: {concept.invalid_reason}")
        synonyms = self.vocabulary.synonyms(concept_id)
        if synonyms:
            lines.append(f"synonyms: {'; '.join(synonyms)}")
        relationships = self.vocabulary.relationships(concept_id)
        related_ids = {
            target
            for rel, target in relationships
            if rel in LOINC_AXES or rel == "Maps to"
        }
        names = self.vocabulary.by_id(related_ids)
        axes = {
            LOINC_AXES[rel]: names[target].concept_name
            for rel, target in relationships
            if rel in LOINC_AXES and target in names
        }
        if axes:
            lines.append(
                "LOINC axes: "
                + "; ".join(
                    f"{axis}={axes[axis]}"
                    for axis in LOINC_AXES.values()
                    if axis in axes
                )
            )
        maps_to = [
            names[target]
            for rel, target in relationships
            if rel == "Maps to" and target in names and target != concept_id
        ]
        for target in maps_to:
            lines.append(f"maps to: {Candidate.from_concept(target).line()}")
        return "\n".join(lines)

    def hierarchy(
        self,
        concept_id: int,
        direction: Literal["parents", "children"],
        domain: Optional[str] = None,
        vocabularies: Optional[Sequence[str]] = None,
        top_k: Optional[int] = None,

        return self.vocabulary.neighbours(
            concept_id, direction, domain, vocabularies, limit=(top_k or self.top_k) * 2

    def select(  # type: ignore[override]
        self,
        results: RetrieverResults,
        contexts: Optional[Sequence[CodingContext]] = None,
        where: Optional[dict[str, Any]] = None,
    ) -> SelectorResults:
        """Select a concept for each query.

        Args:
            results: The retrieved candidates of each query.
            contexts: The context of each query. Defaults to the query alone.
            where: The retriever filter of this call; its domain and vocabularies
                restrict the agent's searches too.
        """

            raise ValueError(
                "AgenticSelector has no retriever or vocabulary: use it in a "
                "TerminologyMapper, or pass retriever=..."
            )
        domain, vocabularies = scope_from_where(where)
        domain = domain or self.domain
        if vocabularies:
            vocabularies = [v for v in vocabularies if v in self.vocabularies]
        else:
            vocabularies = self.vocabularies

            contexts = [CodingContext(text=q) for q in results.queries]
        if len(contexts) != len(results.queries):
            raise ValueError(
                f"Got {len(contexts)} contexts for {len(results.queries)} queries"
            )
        items = [
            (context, candidates_from_expressions(expressions))
            for context, expressions in zip(contexts, results.results)
        ]
        found: list[ConceptSearchResult] = run_sync(
            self.agent.search_many(
                items, domain, vocabularies, max_concurrency=self.max_concurrency
            )
        )
        return SelectorResults(
            results=[
                selection(result, expressions)
                for result, expressions in zip(found, results.results)
            ],
            queries=results.queries,
        )

    def select(  # type: ignore[override]
    self,
    results: RetrieverResults,
    contexts: Optional[Sequence[CodingContext]] = None,
    where: Optional[dict[str, Any]] = None,
) -> SelectorResults:
    """Select a concept for each query.

    Args:
        results: The retrieved candidates of each query.
        contexts: The context of each query. Defaults to the query alone.
        where: The retriever filter of this call; its domain and vocabularies
            restrict the agent's searches too.
    """
    if self.agent is None:
        raise ValueError(
            "AgenticSelector has no retriever or vocabulary: use it in a "
            "TerminologyMapper, or pass retriever=..."
        )
    domain, vocabularies = scope_from_where(where)
    domain = domain or self.domain
    if vocabularies:
        vocabularies = [v for v in vocabularies if v in self.vocabularies]
    else:
        vocabularies = self.vocabularies
    if contexts is None:
        contexts = [CodingContext(text=q) for q in results.queries]
    if len(contexts) != len(results.queries):
        raise ValueError(
            f"Got {len(contexts)} contexts for {len(results.queries)} queries"
        )
    items = [
        (context, candidates_from_expressions(expressions))
        for context, expressions in zip(contexts, results.results)
    ]
    found: list[ConceptSearchResult] = run_sync(
        self.agent.search_many(
            items, domain, vocabularies, max_concurrency=self.max_concurrency
        )
    )
    return SelectorResults(
        results=[
            selection(result, expressions)
            for result, expressions in zip(found, results.results)
        ],
        queries=results.queries,
    )

    def scope_from_where(
    where: Optional[dict[str, Any]],
) -> tuple[Optional[str], Optional[list[str]]]:
    """The domain and vocabularies of a retriever filter built by `retriever_filter`.

    Raises:
        ValueError: For filters on other fields, which the agent's tools cannot apply.
    """
    if not where:
        return None, None
    conditions = where["$and"] if set(where) == {"$and"} else [where]
    domain: Optional[str] = None
    vocabularies: Optional[list[str]] = None
    for condition in conditions:
        for key, value in condition.items():
            if key == "std_domain_id" and isinstance(value, str):
                domain = value
            elif key == "std_domain_id" and isinstance(value, dict) and "$eq" in value:
                domain = value["$eq"]
            elif key == "std_vocabulary_id" and isinstance(value, str):
                vocabularies = [value]
            elif (
                key == "std_vocabulary_id"
                and isinstance(value, dict)
                and "$in" in value
            ):
                vocabularies = list(value["$in"])
            else:
                raise ValueError(
                    f"AgenticSelector understands filters on std_domain_id and "
                    f"std_vocabulary_id only, got {condition!r}"
                )
    return domain, vocabularies

    def source_context(source_concept: Any, text: str) -> CodingContext:
    """The context of a `SourceConcept`: its code and vocabulary, and the original text."""
    related: dict[str, str] = {}
    for field in ("source_code", "source_vocabulary_id"):
        value = getattr(source_concept, field, None)
        if value and str(value).lower() not in ("nan", "none"):
            related[field] = str(value)
    original = getattr(source_concept, "source_code_description", None)
    if original and original != text:
        related["original_text"] = original
    return CodingContext(text=text, related=related)

    def selection(
    result: ConceptSearchResult, expressions: Sequence[Any]
) -> SelectedExpressionMetadata | EmptySelectionMetadata:
    """A selector result from an agent result.

    The retrieved expression of the chosen concept is used when there is one;
    otherwise the concept's name, with ``result_list_index = -1``. The distance is
    ``1 - score``, so the mapper's ``confidence_score`` is the agent's score.
    """
    review: dict[str, Any] = {
        "rationale": result.rationale,
        "trace": json.dumps(result.trace, ensure_ascii=False),
        "usage": result.usage(),
    }
    concept: Optional[Candidate] = result.concept
    if concept is None:
        return EmptySelectionMetadata(**review)
    distance = 1 - result.score if result.score is not None else None
    for index, expression in enumerate(expressions):
        if (
            expression.std_concept_id
            and int(expression.std_concept_id) == concept.concept_id
        ):
            data = expression.model_dump(exclude={"distance", "expression_id"})
            return SelectedExpressionMetadata(
                **data, distance=distance, result_list_index=index, **review
            )
    return SelectedExpressionMetadata(
        expression=concept.concept_name,
        expression_concept_id=str(concept.concept_id),
        expression_origin=ExpressionOrigin.STANDARD_CONCEPT,
        std_concept_id=str(concept.concept_id),
        std_concept_name=concept.concept_name,
        std_vocabulary_id=StandardVocabulary(concept.vocabulary_id),
        std_vocabulary_code=concept.concept_code,
        std_domain_id=concept.domain_id,
        distance=distance,
        result_list_index=-1,
        **review,
    )
